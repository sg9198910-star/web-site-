class Developer {

    String role = "Software Developer";

    }
}

// Java Inheritance: Subclass extending Superclass
class JavaDSASpecialist extends Developer {
    String mainLanguage = "Java & DSA";
    String keyTopic = "OOP (Inheritance)";

    void printSpecialization() {
        // Inherited method call
        displayRole();
        System.out.println("Expertise: " + mainLanguage + " | " + keyTopic);
    }
}
